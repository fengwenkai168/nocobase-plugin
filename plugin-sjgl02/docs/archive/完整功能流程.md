```typescript
// 1. 从选中的字段中检测关联字段
const relationFields = selectedFields.filter(f => 
  f.type === 'belongsTo' || 
  f.type === 'hasMany' || 
  f.type === 'belongsToMany'
);

// 2. 提取关联表信息
const relatedTables = relationFields.map(f => ({
  fieldName: f.name,
  fieldTitle: f.title,
  targetCollection: f.target,
  targetTitle: getCollectionTitle(f.target),
  relationType: f.type,
  estimatedRows: getCollectionCount(f.target), // 预估行数
}));
```
